# 생성자 프로퍼티 승격

프로퍼티를 많이 선언하는 클래스에서는 클래스와 프로퍼티를 각각 선언하고, 그 프로퍼티를 설정할 생성자까지 따로 작성해야 해서 반복이 많아져요. 생성자 프로퍼티 승격을 사용하면 이런 반복을 줄일 수 있어요.

```php
<?php

class Book
{
    private string $isbn;
    private string $doi;
    private string $title;

    public function __construct(string $isbn, string, $doi, string $title)
    {
        $this->isbn = $isbn;
        $this->doi = $doi;
        $this->title = $title;
    }
}

// Using constructor property promotion
class Book
{
    public function __construct(
        private string $isbn,
        private string $doi,
        private string $title
    ) {
    }
}
