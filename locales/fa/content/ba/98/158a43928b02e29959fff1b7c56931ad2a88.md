# مقدمه

## ساختار

`struct`های C# با `class`ها رابطه‌ی نزدیکی دارند. آن‌ها حالت و رفتار دارند. سازنده‌هایی دارند که آرگومان می‌گیرند، می‌توان نمونه‌ها را به متغیرها تخصیص داد، برابری آن‌ها را بررسی کرد و در مجموعه‌ها ذخیره کرد.

```csharp
enum Unit
{
    Kg,
    Lb
}
struct Weight
{
    private double count;
    private Unit unit;

    public Weight(double count, Unit unit)
    {
        this.count = count;
        this.unit = unit;
    }

    public override string ToString()
    {
        return count.ToString() + unit.ToString();
    }
}

new Weight(77.5, Unit.Kg).ToString();
// => "77.6Kg"
```
