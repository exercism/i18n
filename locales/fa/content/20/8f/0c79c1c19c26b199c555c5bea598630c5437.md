# ارتقای ویژگی در سازنده

در کلاس‌هایی که ویژگی‌های زیادی را تعریف می‌کنند، تعریف جداگانه‌ی کلاس، ویژگی‌هایش و یک سازنده برای تنظیم آن ویژگی‌ها نیازمند تکرار است.
با استفاده از ارتقای ویژگی در سازنده می‌توانیم میزان تکرار را کاهش دهیم.

```php
<?php

class Book
{
    private string $isbn;
    private string $doi;
    private string $title;

    public function __construct(string $isbn, string, $doi, string $title)
    {
        $this->isbn = $isbn;
        $this->doi = $doi;
        $this->title = $title;
    }
}

// Using constructor property promotion
class Book
{
    public function __construct(
        private string $isbn,
        private string $doi,
        private string $title
    ) {
    }
}
