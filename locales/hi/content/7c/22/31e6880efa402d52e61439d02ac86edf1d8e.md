# कंस्ट्रक्टर प्रॉपर्टी प्रोमोशन

जब किसी क्लास में बहुत सारी प्रॉपर्टी घोषित करनी हों, तो क्लास, उसकी प्रॉपर्टी और प्रॉपर्टी को वैल्यू देने वाले कंस्ट्रक्टर, इन तीनों को अलग-अलग घोषित करना पड़ता है। इससे बहुत दोहराव होता है। कंस्ट्रक्टर प्रॉपर्टी प्रोमोशन का उपयोग करके हम इस दोहराव को कम कर सकते हैं।

```php
<?php

class Book
{
    private string $isbn;
    private string $doi;
    private string $title;

    public function __construct(string $isbn, string, $doi, string $title)
    {
        $this->isbn = $isbn;
        $this->doi = $doi;
        $this->title = $title;
    }
}

// Using constructor property promotion
class Book
{
    public function __construct(
        private string $isbn,
        private string $doi,
        private string $title
    ) {
    }
}
