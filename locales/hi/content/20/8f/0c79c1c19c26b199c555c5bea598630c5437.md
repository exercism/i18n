# कंस्ट्रक्टर प्रॉपर्टी प्रमोशन

जिन क्लास में बहुत सारी प्रॉपर्टी घोषित होती हैं, उनमें क्लास को, उसकी प्रॉपर्टी को और प्रॉपर्टी को वैल्यू असाइन करने वाले कंस्ट्रक्टर को अलग-अलग घोषित करना पड़ता है। इसमें बहुत दोहराव हो जाता है।

कंस्ट्रक्टर प्रॉपर्टी प्रमोशन का उपयोग करके इस दोहराव को कम किया जा सकता है।

```php
<?php

class Book
{
    private string $isbn;
    private string $doi;
    private string $title;

    public function __construct(string $isbn, string, $doi, string $title)
    {
        $this->isbn = $isbn;
        $this->doi = $doi;
        $this->title = $title;
    }
}

// Using constructor property promotion
class Book
{
    public function __construct(
        private string $isbn,
        private string $doi,
        private string $title
    ) {
    }
}
