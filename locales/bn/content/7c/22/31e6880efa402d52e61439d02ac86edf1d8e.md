# কনস্ট্রাক্টর প্রপার্টি প্রমোশন

যেসব ক্লাসে অনেক প্রপার্টি থাকে, সেখানে ক্লাস, তার প্রপার্টি আর প্রপার্টিগুলো সেট করার জন্য কনস্ট্রাক্টর আলাদা আলাদা ডিক্লেয়ার করলে একই কথা বারবার লিখতে হয়। কনস্ট্রাক্টর প্রপার্টি প্রমোশন ব্যবহার করে এই পুনরাবৃত্তি কমানো যায়।

```php
<?php

class Book
{
    private string $isbn;
    private string $doi;
    private string $title;

    public function __construct(string $isbn, string, $doi, string $title)
    {
        $this->isbn = $isbn;
        $this->doi = $doi;
        $this->title = $title;
    }
}

// Using constructor property promotion
class Book
{
    public function __construct(
        private string $isbn,
        private string $doi,
        private string $title
    ) {
    }
}
