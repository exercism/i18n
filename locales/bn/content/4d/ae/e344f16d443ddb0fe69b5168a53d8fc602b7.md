# পরিচিতি

## কলব্যাক ফাংশন

কলব্যাক ফাংশন হলো এমন ফাংশন, যেগুলো আর্গুমেন্ট হিসেবে পাঠানো হয়।
এই প্রোগ্রামিং প্যাটার্নটি সিঙ্ক্রোনাস ও অ্যাসিঙ্ক্রোনাস দুই ধরনের প্রোগ্রামিংয়েই পরপর একগুচ্ছ ফাংশন কল তৈরি করে।
একটি কলব্যাক ফাংশন লেখা আর একটি ফাংশন লেখা একই ব্যাপার; তবে কলব্যাক ফাংশনটিকে অবশ্যই কল করা ফাংশনটি যে সিগনেচার নির্ধারণ করেছে তার সাথে মিলতে হবে।

```javascript
const squareLength = 5;

// Caller function takes a callback function
function applyToSquare(callback) {
  return callback(squareLength);
}

// Callback must expect the possible argument from the calling function
function areaOfSquare(number) {
  return number * number;
}

applyToSquare(areaOfSquare); // => 25
```

আপনি চাইলে ফাংশন এক্সপ্রেশন, অ্যানোনিমাস ফাংশন এক্সপ্রেশন বা অ্যারো ফাংশন এক্সপ্রেশন হিসেবেও কলব্যাক লিখতে পারেন:

```javascript
applyToSquare(function squarePerimeter(side) {
  return side * 4;
});

applyToSquare(function (side) {
  return side * 4;
});

applyToSquare((side) => {
  return side * 4;
});
```
