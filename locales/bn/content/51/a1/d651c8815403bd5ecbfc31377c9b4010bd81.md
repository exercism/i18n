# পরিচিতি

প্রধান গাণিতিক ও তুলনা অপারেটরগুলোকে আপনার নিজের **abstract**-এর জন্য ব্যবহারের উপযোগী করে তোলা যায়। এটাই অপারেটর ওভারলোডিং নামে পরিচিত।

একটি অপারেটর ওভারলোড করতে `@:op()` মেটাডেটা ক্লাস ফিল্ড ব্যবহার করুন:

```haxe
abstract MyAbstract(String) {
  public inline function new(s:String) {
    this = s;
  }

  @:op(A * B)
  public function repeat(rhs:Int):MyAbstract {
    var s:StringBuf = new StringBuf();
    for (i in 0...rhs)
      s.add(this);
    return new MyAbstract(s.toString());
  }
}

class Main {
  static public function main() {
    var a = new MyAbstract("foo");
    trace(a * 3); // foofoofoo
  }
}
```
