# পরিচিতি

ইনিশিয়ালাইজার হলো বিশেষ ধরনের মেথড, যা দিয়ে একটি এনামের মান তৈরি করা হয়। এদের সংজ্ঞা অনেকটা মেথডের মতোই দেখতে, শুধু এখানে `func` কিওয়ার্ড নেই, কোনো রিটার্ন টাইপ নেই, নামটি অবশ্যই `init` হতে হবে, আর ইনিশিয়ালাইজারটি _অবশ্যই_ এনামের একটি মান `self`-এ অ্যাসাইন করবে। ইনিশিয়ালাইজার কল করা হয় দুইভাবে: ডট নোটেশন দিয়ে, অথবা এনামের নামের সঙ্গে ইনিশিয়ালাইজারের প্যারামিটারগুলো পাস করে।

```swift
enum Coin {
  case heads
  case tails

  init(_ i: Int) {
    if i.isMultiple(of: 2) {
      self = .heads
    } else {
      self = .tails
    }
  }

  func flip() -> Self {
    switch self {
    case .heads: return .tails
    case .tails: return .heads
    }
  }
}

let tails = Coin.init(13)
// .tails
let heads = Coin(0)
// .heads
let anotherTails = heads.flip()
// .tails
```
