# কনস্ট্রাক্টর প্রপার্টি প্রমোশন

যেসব ক্লাসে অনেক প্রপার্টি ডিক্লেয়ার করা হয়, সেখানে ক্লাস, এর প্রপার্টিগুলো এবং প্রপার্টিগুলো সেট করার জন্য একটি কনস্ট্রাক্টর আলাদাভাবে ডিক্লেয়ার করলে পুনরাবৃত্তি করতে হয়।
কনস্ট্রাক্টর প্রপার্টি প্রমোশন ব্যবহার করে পুনরাবৃত্তির পরিমাণ কমানো যায়।

```php
<?php

class Book
{
    private string $isbn;
    private string $doi;
    private string $title;

    public function __construct(string $isbn, string, $doi, string $title)
    {
        $this->isbn = $isbn;
        $this->doi = $doi;
        $this->title = $title;
    }
}

// Using constructor property promotion
class Book
{
    public function __construct(
        private string $isbn,
        private string $doi,
        private string $title
    ) {
    }
}
