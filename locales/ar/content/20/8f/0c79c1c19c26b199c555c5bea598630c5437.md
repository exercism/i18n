# ترقية خصائص المُنشئ

في الأصناف التي تُعرّف العديد من الخصائص، يتطلب تعريف الصنف وخصائصه ومُنشئ لإسناد الخصائص بشكل منفصل تكرارًا.
باستخدام ترقية خصائص المُنشئ يمكننا تقليل مقدار التكرار.

```php
<?php

class Book
{
    private string $isbn;
    private string $doi;
    private string $title;

    public function __construct(string $isbn, string, $doi, string $title)
    {
        $this->isbn = $isbn;
        $this->doi = $doi;
        $this->title = $title;
    }
}

// Using constructor property promotion
class Book
{
    public function __construct(
        private string $isbn,
        private string $doi,
        private string $title
    ) {
    }
}
