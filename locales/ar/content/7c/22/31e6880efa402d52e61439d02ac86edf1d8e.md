# ترقية خصائص المُنشئ

في الأصناف التي تُصرّح بالعديد من الخصائص، يتطلب التصريح المنفصل بالصنف وخصائصه وبمُنشئ يضبط هذه الخصائص قدرًا كبيرًا من التكرار.
وباستخدام ترقية خصائص المُنشئ يمكننا تقليل مقدار هذا التكرار.

```php
<?php

class Book
{
    private string $isbn;
    private string $doi;
    private string $title;

    public function __construct(string $isbn, string, $doi, string $title)
    {
        $this->isbn = $isbn;
        $this->doi = $doi;
        $this->title = $title;
    }
}

// Using constructor property promotion
class Book
{
    public function __construct(
        private string $isbn,
        private string $doi,
        private string $title
    ) {
    }
}
