# مقدمة

## البُنى

ترتبط البُنى `struct` في C# ارتباطًا وثيقًا بالأصناف `class`. لها حالة وسلوك. ولها مُنشِئات تأخذ وسائط، ويمكن إسناد النسخ منها واختبارها للتساوي وتخزينها في مجموعات.

```csharp
enum Unit
{
    Kg,
    Lb
}
struct Weight
{
    private double count;
    private Unit unit;

    public Weight(double count, Unit unit)
    {
        this.count = count;
        this.unit = unit;
    }

    public override string ToString()
    {
        return count.ToString() + unit.ToString();
    }
}

new Weight(77.5, Unit.Kg).ToString();
// => "77.6Kg"
```
